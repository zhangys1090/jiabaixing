    # python/agent/persistence/database.py 新增
from contextlib import asynccontextmanager
from typing import AsyncIterator

@asynccontextmanager
async def database_session() -> AsyncIterator[Any]:
    """统一数据库会话——自动选择 SQLite/PostgreSQL 后端"""
    backend = get_database_backend()
    if backend == "postgres":
        engine = get_async_engine()
        async with engine.begin() as conn:
            yield conn
    else:
        import aiosqlite
        async with aiosqlite.connect(str(SQLITE_DEFAULT_PATH)) as conn:
            conn.row_factory = aiosqlite.Row
            yield conn
  2. 改造 session_store.py——将同步 get_sync_connection() 替换为 async with database_session() as conn

Phase 2 AST 自动发现引擎（1天）

新增 python/agent/tools/auto_discovery.py： 

import ast
import importlib
import os
from pathlib import Path
from typing import Any

class ToolAutoDiscovery:
    """AST 扫描工具目录，自动发现并注册工具。

    约定：
    1. 工具文件位于 tools/ 目录下，文件名匹配 *_tools.py 或 tool_*.py
    2. 文件导出 TOOL_DEFINITION (ToolDefinition) 和 TOOL_EXECUTOR (callable)
    3. 可通过 AST 静态扫描确认文件包含 __all__ 或 TOOL_DEFINITION 赋值
    """

    TOOLS_DIR = Path(__file__).parent

    def discover(self, registry: "ToolRegistry") -> int:
        """扫描 tools/ 目录，自动注册所有符合约定的工具。"""
        count = 0
        for py_file in self._find_tool_files():
            if self._has_tool_exports(py_file):
                module = self._import_module(py_file)
                if hasattr(module, "TOOL_DEFINITION") and hasattr(module, "TOOL_EXECUTOR"):
                    registry.register(module.TOOL_DEFINITION, module.TOOL_EXECUTOR)
                    count += 1
        return count

    def _find_tool_files(self) -> list[Path]:
        """发现工具文件：*_tools.py 或 tool_*.py，排除 __init__.py 和 registry.py"""
        ...

    def _has_tool_exports(self, path: Path) -> bool:
        """AST 静态检查：文件是否包含 TOOL_DEFINITION 赋值"""
        tree = ast.parse(path.read_text(encoding="utf-8"))
        for node in ast.walk(tree):
            if isinstance(node, ast.Assign):
                for target in node.targets:
                    if isinstance(target, ast.Name) and target.id == "TOOL_DEFINITION":
                        return True
        return False

Phase 1 平台定义（0.5天）

from enum import Enum

class AgentPlatform(str, Enum):
    """Agent 接入平台——不同平台有不同的工具权限边界"""
    CLI = "cli"           # 命令行：全功能
    FEISHU = "feishu"     # 飞书：IM限制（无桌面/浏览器）
    WECHAT = "wechat"     # 微信：IM限制 + 消息长度限制
    DINGTALK = "dingtalk" # 钉钉：IM限制
    HTTP_API = "http_api" # OpenAI兼容API：全功能但需审批
    WEB_UI = "web_ui"     # 前端Web界面：全功能
    CRON = "cron"         # 定时任务：仅系统/网络/记忆

PLATFORM_TOOLSET_MAP: dict[AgentPlatform, str] = {
    AgentPlatform.CLI: "full",
    AgentPlatform.FEISHU: "daily",       # IM场景：日常+记忆+网络搜索
    AgentPlatform.WECHAT: "daily",
    AgentPlatform.DINGTALK: "daily",
    AgentPlatform.HTTP_API: "full",
    AgentPlatform.WEB_UI: "full",
    AgentPlatform.CRON: "minimal",       # 定时任务：最小工具集
}

PLATFORM_RISK_CONSTRAINTS: dict[AgentPlatform, list[str]] = {
    AgentPlatform.FEISHU: ["shell_exec", "code_generate", "desktop_automate"],
    AgentPlatform.WECHAT: ["shell_exec", "code_generate", "desktop_automate"],
    # CLI 无限制
}

Phase 2 集成到 ToolRegistry（0.5天）

在 ToolRegistry 新增 resolve_for_platform() 方法： 

def resolve_for_platform(
    self, platform: AgentPlatform, toolset_registry: "ToolsetRegistry"
) -> list[ToolDefinition]:
    """根据平台解析可用工具集——组合平台工具集映射 + 风险约束"""
    toolset_id = PLATFORM_TOOLSET_MAP.get(platform, "full")
    resolved = toolset_registry.resolve(toolset_id, self)
    excluded = set(PLATFORM_RISK_CONSTRAINTS.get(platform, []))
    return [t for t in resolved.tool_names if t not in excluded]
验收标准：
  • 飞书平台调用时，shell_exec/desktop_automate 等危险工具不可用
  • CLI 平台全功能可用
  • 新增平台只需在 PLATFORM_TOOLSET_MAP 加一行
